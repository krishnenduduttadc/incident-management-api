//Configure systemd to run Flask automatically
sudo nano /etc/systemd/system/incident-api.service
//Add
[Unit]
Description=Incident Management Flask API
After=network.target postgresql.service
Requires=postgresql.service

[Service]
User=incidentapp
Group=incidentapp

WorkingDirectory=/opt/incident-management-api

EnvironmentFile=/etc/incident-management/incident-api.env

ExecStart=/opt/incident-management-api/.venv/bin/gunicorn \
    --workers 3 \
    --bind 127.0.0.1:8000 \
    --access-logfile - \
    --error-logfile - \
    --timeout 60 \
    run:app

Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

//Configure Nginx as a reverse proxy(Gunicorn is listening only on 127.0.0.1:8000. Nginx will expose the API to the network.)
sudo nano /etc/nginx/sites-available/incident-api
//Add
server {
    listen 80;
    server_name YOUR_SERVER_IP;

    client_max_body_size 10M;

    location / {
        proxy_pass http://127.0.0.1:8000;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
}

//Enable configuration
sudo ln -s \
    /etc/nginx/sites-available/incident-api \
    /etc/nginx/sites-enabled/incident-api

sudo ln -s \
    /etc/nginx/sites-available/incident-api \
    /etc/nginx/sites-enabled/incident-api

sudo nginx -t

//Restart nginx

sudo systemctl restart nginx
